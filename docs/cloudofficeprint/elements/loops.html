 91class ForEachSheet(ForEach):
 92    """Loop where a sheet will be repeated for each element of the loop. Only supported in Excel templates."""
 93
 94    def __init__(self, name: str, content: Union[Iterable[Element], Mapping[str, Element]]):
 95        """
 96        Args:
 97            name (str): The name for this element (Cloud Office Print tag).
 98            content (Union[Iterable[Element], Mapping[str, Element]]): An iterable or mapping containing the elements for this loop element.
 99        """
100        # when content is a mapping, it means "sheet name": content for that sheet
101        # when it's just an iterable, don't add sheet names (or users can add a Property manually)
102        if isinstance(content, Mapping):
103            new_content = []
104            for sheetname, sheetcontent in content.items():
105                # we need to add the additional sheet_name property,
106                # so we should convert the Element to an ElementCollection if needed
107                if not isinstance(sheetcontent, ElementCollection):
108                    sheetcontent = ElementCollection.element_to_element_collection(
109                        sheetcontent)
110                # adding the new property containing sheet_name
111                sheetcontent.add(Property("sheet_name", sheetname))
112                new_content.append(sheetcontent)
113            content = new_content
114        # at this point, content is always Iterable[Element]
115
116        super().__init__(name, content)
117        self._tags = {
118            "{!" + name + "}"
119        }

132        """

134        self._tags = {
135            "{:" + name + "}",
136            "{/" + name + "}"
137        }
138        self._distribute = distribute
139    @property

141        data = super().as_dict
142        if self._distribute:
143            data[f"{self.name}_distribute"] = True
