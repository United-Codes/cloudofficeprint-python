  6class Code(Element, ABC):
  7    """The abstract base class for QR-codes and barcodes"""
  8
  9    def __init__(self, name: str, data: str, type: str):
 10        """
 11        Args:
 12            name (str): The name for this Code object (Cloud Office Print tag).
 13            data (str): The data for this Code object.
 14            type (str): For the different types of QR-codes and barcodes, we refer to the Cloud Office Print documentation.
 15        """
 16        super().__init__(name)
 17        self.data: str = data
 18        self.type: str = type
 19
 20    @property
 21    def available_tags(self) -> FrozenSet[str]:
 22        return frozenset({"{|" + self.name + "}"})
 23
 24    @property
 25    def _dict_suffixes(self) -> Dict:
 26        """Get the suffixes that need to be appended to the keys of the dict representation of this Code object.
 27
 28        Returns:
 29            Dict: the suffixes that need to be appended to the keys of the dict representation of this Code object
 30        """
 31        result = {}
 32
 33        if self.type is not None:
 34            result["_type"] = self.type
 35
 36        return result
 37
 38    @property
 39    def as_dict(self) -> Dict:
 40        result = {
 41            self.name: self.data
 42        }
 43
 44        for suffix, value in self._dict_suffixes.items():
 45            result[self.name + suffix] = value
 46
 47        return result
