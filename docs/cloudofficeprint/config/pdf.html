369    def set_page_margin_at(self, value: int, position: str = None):
370        """Set page_margin
371
372        Either set the position for all margin positions (if position is None) or set a specific one.
373
374        Args:
375            value (int): page margin in px
376            position (str, optional): "all", "top", "bottom", "left" or "right". Defaults to None.
377        """
378        if position is not None:
379            if isinstance(self.page_margin, Mapping):
380                # page margin is already a dict, add/change this position
381                self.page_margin[position] = value
382            elif self.page_margin is None:
383                # page margin not yet defined, set it to a dict with this position defined
384                self.page_margin = {position: value}
385            else:
386                # page margin defined but no dict, convert to dict first
387                current = self.page_margin
388                self.page_margin = {
389                    "top": current,
390                    "bottom": current,
391                    "left": current,
392                    "right": current,
393                }
394                self.page_margin[position] = value
395        else:
396            self.page_margin = value
